              <Text style={styles.productDesc} numberOfLines={2}>{item.description}</Text>

              )}
              {item.materialBlend && (
                <Text style={styles.specialSpec}>Material: {item.materialBlend}</Text>
              )}
            </View>
            <View style={styles.cardAction}>
              <Text style={styles.productPrice}>${item.price.toFixed(2)}</Text>
              <TouchableOpacity style={styles.miniCartButton}>
                <Text style={styles.miniCartButtonText}>+</Text>
              </TouchableOpacity>
            </View>
          </View>
